Public Class Student

    Public Grade1 As Double
    Public Grade2 As Double
    Public Grade3 As Double

    Public Function Average() As Double
        Return (Grade1 + Grade2 + Grade3) / 3
    End Function

    Public Function PassOrFail() As String
        If Average() >= 10 Then        ' ← conditional INSIDE the class
            Return "Επιτυχία ✓"
        Else
            Return "Αποτυχία ✗"
        End If
    End Function

End Class



Public Class Product
    Public Price As Double
    Public VatRate As Double

    Public Function VatAmount() As Double
        Return Price * (VatRate / 100)
    End Function

    Public Function FinalPrice() As Double
        Return Price + VatAmount()
    End Function
End Class

' In Form1:
Private Sub btnCalculate_Click(...) Handles btnCalculate.Click
    Dim p As New Product()
    p.Price = Double.Parse(txtPrice.Text)
    p.VatRate = Double.Parse(txtVat.Text)

    lblResult.Text = "ΦΠΑ: " & p.VatAmount().ToString("F2") & " €" &
                     Environment.NewLine &
                     "Τελική Τιμή: " & p.FinalPrice().ToString("F2") & " €"
End Sub


Public Class Form1
    Private Sub btnCalculate_Click(sender As Object, e As EventArgs) Handles btnCalculate.Click
        Dim length As Double
        Dim width As Double

        If Double.TryParse(txtLength.Text, length) Then
            ' valid
        Else
            MessageBox.Show("Παρακαλώ εισάγετε έγκυρο μήκος!")
            Exit Sub
        End If

        If Double.TryParse(txtWidth.Text, width) Then
            ' valid
        Else
            MessageBox.Show("Παρακαλώ εισάγετε έγκυρο πλάτος!")
            Exit Sub
        End If

        Dim area As Double = length * width

        lblResult.Text = "Εμβαδόν: " & area.ToString("F2") & " m²"
    End Sub
End Class


Public Class Form1
    Private Sub btnConvert_Click(sender As Object, e As EventArgs) Handles btnConvert.Click
        Dim celsius As Double

        If Double.TryParse(txtCelsius.Text, celsius) Then
            ' valid
        Else
            MessageBox.Show("Παρακαλώ εισάγετε έγκυρη θερμοκρασία!")
            Exit Sub
        End If

        Dim fahrenheit As Double = (celsius * 9 / 5) + 32

        lblResult.Text = celsius.ToString("F2") & " °C = " & fahrenheit.ToString("F2") & " °F"
    End Sub
End Class
